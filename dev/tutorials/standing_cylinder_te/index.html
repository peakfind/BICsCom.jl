We sweep the frequency

ks = 0.4:1e-5:1.0for k in ks    cydc = build_cylinder_cache(4n, k, r, inn, ext)    coef = get_coeff(4n, cydc, inn, ext)    bydc = build_boundary_cache(sp, 4n, k, ext)    Λ, _ = assemble_dtn(sp, 4n, coef, bydc, k, ext)    Δ = apply_bc(Λ, n)    apply_tbc!(Δ, sp, k; homo = hom)    s = svdvals(Δ)    push!(msv, minimum(s))end
